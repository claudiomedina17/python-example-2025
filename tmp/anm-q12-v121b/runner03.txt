            except torch.OutOfMemoryError as exc:
                state['outcome']='EXECUTION_INVALID';state['stop_reason']='OOM_AFTER_EXPOSURE';persist('PA',idx,{'probe_id':item['probe_id'],'source_item_id':source,'classification':'OOM','raw':None,'parsed':None,'error':repr(exc),'semantic_score':None});break

            state['pa']['attempted']+=1;state['pa']['valid']+=1

            if any(v>2 for v in state['pa']['class_errors'].values()): state['outcome']='NOT_QUALIFIED';state['stop_reason']='QG2_DIRECTIONAL_CAP';break
        if state['outcome'] is not None:return final_post()
