def post_paste(slug,payload,attempts=4):
    body={
      'title':slug,'custom_slug':slug,'visibility':'unlisted','expiry':'1h',
      'language':'json','content':json.dumps(payload,ensure_ascii=False,indent=2)
    }
    last=None
    for k in range(attempts):
        try:
            r=requests.post('https://pastepile.com/api/public/pastes',headers={'Content-Type':'application/json'},json=body,timeout=30)
            if 200 <= r.status_code < 300:
                return {'status_code':r.status_code,'text':r.text[:300]}
            last=f'HTTP {r.status_code}: {r.text[:300]}'
        except Exception as exc:
            last=repr(exc)
        time.sleep(2*(k+1))
    raise RuntimeError(f'PERSISTENCE_FAILURE {slug}: {last}')


def raw_url(path):
    return f'https://raw.githubusercontent.com/claudiomedina17/python-example-2025/main/{path}'


def fetch_bundle(paths_csv,key):
    parts=[]
    for path in [x.strip() for x in paths_csv.split(',') if x.strip()]:
        r=requests.get(raw_url(path),timeout=60)
        r.raise_for_status()
        parts.append(r.text.strip())
    if not parts:
        raise RuntimeError('EMPTY_BUNDLE_PATH')
    cipher=base64.b64decode(''.join(parts))
    compressed=Fernet(key.encode()).decrypt(cipher)
    plain=zlib.decompress(compressed)
    return json.loads(plain.decode('utf-8'))


def render_item(item):
    return 'ITEM:\n'+item['stem']+'\n\nOPTIONS:\n'+'\n'.join(f'{k}. {v}' for k,v in item['options'].items())

def parse_generated(generated,adapter):
    if isinstance(generated,BaseModel):
        obj=generated.model_dump(); raw=generated.model_dump_json()
    elif isinstance(generated,str):
        raw=generated; obj=json.loads(raw)
    else:
        raw=str(generated); obj=json.loads(raw)
    validated=adapter.validate_python(obj)
    return raw,validated.model_dump()


def context_candidates(cfg,tok):
    vals=[]
    for obj in [cfg,getattr(cfg,'text_config',None)]:
        if obj is None: continue
        for field in ['max_position_embeddings','n_positions','max_sequence_length','seq_length']:
            v=getattr(obj,field,None)
            if isinstance(v,int) and 0<v<10**9:
                vals.append({'source':f'config.{field}','value':v})
    tv=getattr(tok,'model_max_length',None)
    if isinstance(tv,int) and 0<tv<10**9:
        vals.append({'source':'tokenizer.model_max_length','value':tv})
    return vals
